cmake_minimum_required(VERSION 3.21)
project(NEXVARY_MR VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_library(mr_core cpp/src/RepairCore.cpp)
target_include_directories(mr_core PUBLIC cpp/include)
if(MSVC)
  target_compile_options(mr_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(mr_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(mr_core_tests cpp/tests/core_tests.cpp)
  target_link_libraries(mr_core_tests PRIVATE mr_core)
  add_test(NAME mr_core_tests COMMAND mr_core_tests)
endif()
option(MR_BUILD_GUI "Build Qt 6 Quick UI" OFF)
if(MR_BUILD_GUI)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Quick QuickControls2)
  qt_standard_project_setup(REQUIRES 6.5)
  qt_add_executable(nexvary_mr cpp/src/main.cpp cpp/src/ToolBridge.cpp cpp/src/ToolBridge.hpp cpp/src/RepairBridge.cpp cpp/src/RepairBridge.hpp)
  qt_add_qml_module(nexvary_mr URI NexvaryMR VERSION 1.0 QML_FILES qml/Main.qml)
  target_link_libraries(nexvary_mr PRIVATE mr_core Qt6::Quick Qt6::QuickControls2)
  set_target_properties(nexvary_mr PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
